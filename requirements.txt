# The demo runs with no dependencies at all - server, NLU, retrieval and the
# extractive fallback are stdlib only. Install ONE of the lines below to switch
# the generative layer on; the adapter picks it up automatically.
#
#   anthropic>=0.40        # LLM_PROVIDER=anthropic   ANTHROPIC_API_KEY
#   openai>=1.40           # LLM_PROVIDER=openai      OPENAI_API_KEY
#   openai>=1.40           # LLM_PROVIDER=groq        GROQ_API_KEY   (Groq speaks
#                          #   the OpenAI wire format, so it needs no extra package)
#   google-genai>=0.3      # LLM_PROVIDER=gemini      GEMINI_API_KEY
#
# Groq has the most usable free tier - see README § Running the demo for free.
#
# Installing more than one is fine - LLM_PROVIDER decides which is used, and
# with LLM_PROVIDER=auto the first configured one in PROVIDER_ORDER wins.

anthropic>=0.40
openai>=1.40

# Required (not optional like the LLM providers above) - server.py loads .env
# at startup before any app imports so provider keys are picked up.
python-dotenv>=1.0

# Voice (app/tts.py) - optional, independent of the LLM providers above:
#
#   gtts>=2.5                                # TTS_PROVIDER=gtts (free, no key,
#                                             #   also what pregenerate_voice.py
#                                             #   defaults to - see data/voice_cache/)
#   VBEE_TTS_KEY + VBEE_APP_ID env vars only  # TTS_PROVIDER=vbee - no package,
#                                             #   called over plain HTTP
#
# Without either, /api/tts reports unavailable and the mic/speaker controls
# hide themselves - the rest of the demo is unaffected.
gtts>=2.5
